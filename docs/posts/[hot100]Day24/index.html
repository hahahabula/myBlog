1 最长有效括号

分析

用动态规划的方式进行求解：

  1. 首先所有以左括号结束的子串都不合法
  2. 以右括号结尾的子串的最大合法长度有三个部分合成：
    • 前面与该右括号配对的左括号，同时该左括号必须在上图 Figure 1 的箭头处，否则无法继续往前，此时最大长度为0
    • 若在箭头处，则最大的长度为红色括号中合法子串的长度+2+蓝色部分的长度

  • C++
class Solution:
    def longestValidParentheses(self, s: str) -> int:
        if len(s) == 0:  # 当字符串长度为0时最长就是0
            return 0
        dp = [0] * len(s)
        # 状态转移方程dp[i] = dp[i-1] + dp[i-dp[i-1]-1] + 2
        for i in range(len(s)):
            if s[i] == '(':
                continue
            if i-1 >= 0:
                if i-dp[i-1]-1 >= 0 and s[i-dp[i-1]-1] == '(':
                    dp[i] += dp[i-1] + 2 + dp[i-dp[i-1]-2]
        return max(dp)

2 柱状图中的最大矩形

分析

采用单调栈：

单调增栈可用于搜寻某处左右两边比其小的元素，构建左右边界。

  • Python
  • C++
class Solution:
    def largestRectangleArea(self, heights: List[int]) -> int:
        stack = []
        heights.append(0)  # 用于额外界定边界
        max_zone = 0; i = 0
        while True: 
            if len(stack) == 0 or heights[i] > heights[stack[-1]]:
                stack.append(i)
                i += 1
            else:
                top = stack[-1]
                stack.pop()
                max_zone = max(max_zone, heights[top] * (i - stack[-1] - 1) if len(stack)!= 0 else heights[top] * i)
            if i == len(heights):
                break
        return max_zone
class Solution {
public:
    int largestRectangleArea(vector<int>& height) {
        stack<int> s;
        height.push_back(0);
        int result = 0;
        for (int i = 0; i < height.size();) {
            if (s.empty() || height[i] > height[s.top()])
                s.push(i++);
            else {
                int tmp = s.top();
                s.pop();
                result = max(result,
                             height[tmp] * (s.empty() ? i : i - s.top() - 1));
            }
        }
        return result;
    }
};

4 二叉树的前序遍历

分析

二叉树的三种遍历方式都可以由莫里斯遍历解决，只需要控制输出的顺序：

  1. 莫里斯遍历

莫里斯遍历:[1,2,3,2,4,5,1,6,7,8]

  1. 前序遍历（root->left->right）

在上图 Figure 2 的例子中前序遍历的结果是:[1,2,3,4,5,6,7,8]

对比一下，是否容易得到，前序遍历就是在莫里斯遍历的基础上不打印重复到达节点的第二次遍历。

  • Python
  • C++
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def preorderTraversal(self, root: Optional[TreeNode]) -> List[int]:
        # Mirrors序
        cur = root
        result = []
        mostright = None
        while cur:
            # 指针左子树存在
            if cur.left:
                # 设置前驱节点
                mostright = cur.left
                while mostright.right and mostright.right != cur:
                    mostright = mostright.right
                if not mostright.right:
                    mostright.right = cur
                    result.append(cur.val)
                    cur = cur.left
                    continue
                else:
                    mostright.right = None
                    cur = cur.right
            else:
                result.append(cur.val)
                cur = cur.right
        return result        
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    vector<int> preorderTraversal(TreeNode* root) {
        TreeNode *cur = root;
        vector<int> result;
        TreeNode *mostright = nullptr;
        while (cur) {
            if (cur->left) {
                mostright = cur->left;
                while (mostright->right && mostright->right != cur) {
                    mostright = mostright->right;
                }
                if (!mostright->right) {
                    mostright->right = cur;
                    result.push_back(cur->val);
                    cur = cur->left;
                    continue;
                } else {
                    mostright->right = nullptr;
                    cur = cur->right;
                }
            } else {
                result.push_back(cur->val);
                cur = cur->right;
            }
        }
        return result;
    }
};

中序遍历(left-root-right)的结果是:[3,2,4,5,1,6,7,8]

6 二叉树的后序遍历

分析

二叉树的后序遍历是left->right->root，即：[3,5,4,2,8,7,6,1]

可以观察到后序遍历的结果就是将 Figure 3 中的三条链表从左到右反向输出，反向输出的时机处在：当第二次遍历到会重复遍历的节点时，将其左节点至其最右叶节点的链表反向输出(next=right)，同时最后将(root->(next=right))翻转输出即可得后序遍历。

  • Python
  • C++
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def postorderTraversal(self, root: Optional[TreeNode]) -> List[int]:
        def reverse(dummy: Optional[TreeNode]) -> None:
            pre = dummy.right
            cur = pre.right
            while cur:
                pre.right = cur.right
                cur.right = dummy.right
                dummy.right = cur
                cur = pre.right
        def visit(node: Optional[TreeNode], result: List[int]) -> None:
            dummy = TreeNode(right=node)
            reverse(dummy)
            begin = dummy.right
            while begin:
                result.append(begin.val)
                begin = begin.right
            reverse(dummy)
        
        result = []
        cur = root
        mostright = None
        if not root:
            return result
        # 防止最后一条链失效
        while cur:
            if cur.left:
                mostright = cur.left
                while mostright.right and mostright.right != cur:
                    mostright = mostright.right
                if not mostright.right:
                    mostright.right = cur
                    cur = cur.left
                    continue
                else:
                    mostright.right = None
                    visit(cur.left, result)
                    cur = cur.right
            else:
                cur = cur.right
        visit(root, result)
        return result
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    vector<int> postorderTraversal(TreeNode* root) {
        TreeNode *cur = root;
        vector<int> result;
        TreeNode *mostright = nullptr;
        if (!root) return result;
        while (cur) {
            if (cur->left) {
                mostright = cur->left;
                while (mostright->right && mostright->right != cur) {
                    mostright = mostright->right;
                }
                if (!mostright->right) {
                    mostright->right = cur;
                    cur = cur->left;
                    continue;
                } else {
                    mostright->right = nullptr;
                    visit(cur->left, result);
                    cur = cur->right;
                }
            } else {
                cur = cur->right;
            }
        }
        visit(root, result);  // 最后的根节点到右边界
        return result;
    }
private:
    void reverse(TreeNode& dummy) {
        TreeNode *pre = dummy.right;
        TreeNode *cur = pre->right;
        while (cur) {
            pre->right = cur->right;
            cur->right = dummy.right;
            dummy.right = cur;
            cur = pre->right;
        }
    }
    void visit(TreeNode* node, vector<int>& result) {
        TreeNode dummy{-1, nullptr, node};
        reverse(dummy);
        TreeNode *begin = dummy.right;
        while (begin) {
            result.push_back(begin->val);
            begin = begin->right;
        }
        reverse(dummy);
    }
};
