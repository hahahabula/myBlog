1 有序链表转换二叉搜索树

分析

采用快慢指针法找到每个子链表的中位数，而后将中位数作为根节点，递归地构建左右子树。

  • Python
  • C++
class Solution:
    def sortedListToBST(self, head: Optional[ListNode]) -> Optional[TreeNode]:
        def find_median(head: Optional[ListNode], end: Optional[ListNode]):
            slow = fast = head
            while fast!=end and fast.next!=end:
                slow = slow.next
                fast = fast.next.next
            return slow
        def sortedListToBST(head: Optional[ListNode], end: Optional[ListNode]) -> Optional[TreeNode]:
            if head == end:
                return None
            median = find_median(head, end)
            root = TreeNode(median.val)
            root.left = sortedListToBST(head, median)
            root.right = sortedListToBST(median.next, end)
            return root
        return sortedListToBST(head, None)
class Solution {
public:
    TreeNode* sortedListToBST(ListNode* head) {
        return sortedListToBST(head, nullptr);
    }
    // 利用快慢指针来寻找序列的中位元素
    ListNode* findMedian(ListNode *head, ListNode *end) {
        ListNode *slow = head, *fast = head;
        while (fast != end && fast->next != end) {
            slow = slow->next;
            fast = fast->next->next;
        }
        return slow;
    }  

    TreeNode *sortedListToBST(ListNode *head, ListNode *end) {
        if (head == end) {
            return nullptr;
        }
        ListNode *median = findMedian(head, end);
        TreeNode *root = new TreeNode(median->val);
        root->left = sortedListToBST(head, median);
        root->right = sortedListToBST(median->next, end);
        return root;
    }
};

2 二叉树最小深度

分析
  • 注意：一个节点的存在左或者右节点时，不能说明它便达到了叶节点，需要同时判断左右两边是否都无。
  • 这道题仍采用递归的方式不，当达到叶节点时便返回自己的深度，进行比较
  • Python
  • C++
class Solution:
    def minDepth(self, root: Optional[TreeNode]) -> int:
        # 若一个节点的存在左或者右节点时，不能说明它便达到了叶节点，需要同时判断左右两边是否都无
        INT_MAX = 10**6
        def minDepth(root: Optional[TreeNode], leaf: bool) -> int:
            if not root:
                return 0 if leaf else INT_MAX
            left_depth = minDepth(root.left, not root.right)
            right_depth = minDepth(root.right, not root.left)
            return min(1+left_depth, 1+right_depth)
        return minDepth(root, True)
#include<climits>
class Solution {
public:
    int minDepth(TreeNode* root) {
        return minDepth(root, true);
    }

    int minDepth(TreeNode *root, bool leaf) {
        if (!root) {
            return leaf ? 0 : INT_MAX-1;
        }
        int left_depth = minDepth(root->left, !root->right);
        int right_depth = minDepth(root->right, !root->left);
        return min(left_depth+1, right_depth+1);
    }
};

3 二叉树的最大深度

分析
  • 这道题与二叉树最小深度类似，采用递归的方式，当达到叶节点时便返回自己的深度，进行比较，但该题判断叶节点的方式更为简单。
  • Python
  • C++
class Solution:
    def maxDepth(self, root: Optional[TreeNode]) -> int:
        def maxDepth(root: Optional[TreeNode]) -> int:
            if not root:
                return 0
            left_depth = maxDepth(root.left)
            right_depth = maxDepth(root.right)
            return max(1+left_depth, 1+right_depth)
        return maxDepth(root)
class Solution {
public:
    int maxDepth(TreeNode* root) {
        if (!root) {
            return 0;
        }
        int left_depth = maxDepth(root->left);
        int right_depth = maxDepth(root->right);
        return max(1+left_depth, 1+right_depth);
    }
};

4 路径总和

分析
  • 这道题采用递归的方式，当达到叶节点时，判断当前路径的和是否等于目标值，若等于则返回True，否则返回False。
  • 只有当都false时，才返回False。
  • Python
  • C++
class Solution:
    def hasPathSum(self, root: Optional[TreeNode], targetSum: int) -> bool:
        if not root:
            return False
        if (not root.left and not root.right):
            return targetSum == root.val
        return self.hasPathSum(root.left, targetSum - root.val) or self.hasPathSum(root.right, targetSum - root.val)
class Solution {
public:
    bool hasPathSum(TreeNode* root, int targetSum) {
        if (!root) return false;
        if (!root->left && !root->right) return targetSum == root->val;
        return hasPathSum(root->left, targetSum - root->val) || hasPathSum(root->right, targetSum - root->val);
    }
};

5 路径总和 II

分析

采用递归法，当达到叶节点且路径和等于目标值时，将路径加入结果列表中。而后将当前节点从路径中移除，返回上一层。

  • Python
  • C++
class Solution:
    def pathSum(self, root: Optional[TreeNode], targetSum: int) -> List[List[int]]:
        result = []
        cur = []
        def pathSum(root: Optional[TreeNode], gap: int) -> None:
            if not root:
                return None
            cur.append(root.val)
            if not root.left and not root.right:
                if gap == root.val:
                    result.append(cur[:])  # 若直接.append(cur)会导致元素中是浅拷贝
            pathSum(root.left, gap - root.val)
            pathSum(root.right, gap - root.val)
            cur.pop()
        pathSum(root, targetSum)
        return result
class Solution {
public:
    vector<vector<int>> pathSum(TreeNode* root, int targetSum) {
        vector<vector<int>> result;
        vector<int> cur;
        pathSum(root, targetSum, cur, result);
        return result;
    }

    void pathSum(TreeNode* root, int gap, vector<int> &cur, vector<vector<int>> &result) {
        if (root == nullptr) {
            return;
        }
        cur.push_back(root->val);
        if (!root->left && !root->right) {
            if (gap == root->val) {
                result.push_back(cur);
            }
        }
        pathSum(root->left, gap-root->val, cur, result);
        pathSum(root->right, gap-root->val, cur, result);
        cur.pop_back();
    }
};

6 二叉树中最大路径和

分析
  • 这道题采用递归法，计算当前节点的最大路径和，可以包含左右节点，但最大贡献度不能同时包括左右节点。因为对于某节点，他的贡献只能带其中一个孩子（带大孩子走），整个路径才能向上延伸。
  • 在递归过程中，需要记录最大路径和，并更新。
  • Python
  • C++
class Solution:
    def __init__(self):
        self.maxNumber = float("-inf")
    # 计算当前节点的 最大路径和 可以包含左右节点，但最大贡献度不能同时包括左右节点。因为对于某节点，他的贡献只能带其中一个孩子（带大孩子走），整个路径才能向上延伸
    def maxPathSum(self, root: Optional[TreeNode]) -> int:
        def dfs(root: Optional[TreeNode]) -> int:
            if not root:
                return 0
            left_number = max(dfs(root.left), 0)
            right_number = max(dfs(root.right), 0)
            self.maxNumber = max(self.maxNumber, root.val + left_number + right_number)
            return max(root.val, root.val + max(left_number, right_number))
        
        dfs(root)
        return self.maxNumber
class Solution {
public:
    int maxNumber = INT_MIN;
    int dfs(TreeNode *node) {
        if (!node) return 0;
        int left_number = max(dfs(node->left), 0);
        int right_number = max(dfs(node->right), 0);
        this->maxNumber = max(this->maxNumber, node->val + left_number + right_number);
        return max(node->val, node->val + max(left_number, right_number));   
    }
    int maxPathSum(TreeNode* root) {
        dfs(root);
        return this->maxNumber;
    }
};

7 填充每个节点的下一个右侧节点指针

分析
  • 这道题采用迭代法，使用一个哑节点来连接当前层的节点。
  • 每次遍历当前层的节点，将他们的左右节点连接起来。
  • 然后更新当前层的节点，继续遍历。
  • 直到当前层为空，进入下一层。
  • Python
  • C++
class Solution:
    def connect(self, root: 'Optional[Node]') -> 'Optional[Node]':
        cur = root
        while cur:
            dummy = Node(0)
            pre = dummy
            while cur:
                if cur.left:
                    pre.next = cur.left
                    pre = pre.next
                if cur.right:
                    pre.next = cur.right
                    pre = pre.next
                cur = cur.next
            cur = dummy.next
        return root
class Solution {
public:
    Node* connect(Node* root) {
        Node *cur = root;
        while (cur) {
            Node *dummy = new Node(0);
            Node *pre = dummy;
            while (cur) {
                if (cur->left) {
                    pre->next = cur->left;
                    pre = pre->next;
                }
                if (cur->right) {
                    pre->next = cur->right;
                    pre = pre->next;
                }
                cur = cur->next;
            }
            cur = dummy->next;
        }
        return root;
    }
};

8 求根节点到叶节点数字之和分析

分析
  • 这道题采用深度优先搜索，将当前节点的值加入到路径中，然后递归地遍历左右子树。
  • 当到达叶节点时，将路径中的数字转换为整数，并加入到结果中。
  • 最后返回结果。
  • Python
  • C++
class Solution:
    def __init__(self):
        self.result = 0

    def sumNumbers(self, root: Optional[TreeNode]) -> int:
        # 深度优先
        def dfs(node: Optional[TreeNode], num: str):
            num = num + str(node.val)
            if node.left:
                dfs(node.left, num)
            if node.right:
                dfs(node.right, num)
            if not node.left and not node.right:
                self.result += int(num)
        if not root:
            return 0
        dfs(root, "")
        return self.result
class Solution {
public:
    int result = 0;
    int sumNumbers(TreeNode* root) {
        if (!root) return 0;
        dfs(root, "");
        return this->result;
    }
    void dfs(TreeNode *node, string num) {
        num = num + to_string(node->val);
        if (node->left) {
            dfs(node->left, num);
        }
        if (node->right) {
            dfs(node->right, num);
        }
        if (!node->left && !node->right) {
            this->result += stoi(num);
        }
    }
};
