1 合并两个有序数组

分析

采用双指针法，从后往前遍历，将较大的元素放在nums1的末尾。

  • Python
  • C++
class Solution:
    def merge(self, nums1: List[int], m: int, nums2: List[int], n: int) -> None:
        """
        Do not return anything, modify nums1 in-place instead.
        """
        guard = 10**9 + 1
        nums1_copy = nums1[:m]
        nums1_copy.append(guard); nums2.append(guard)
        i = j = 0
        while  not (nums1_copy[i] == guard and nums2[j] == guard):
            if nums1_copy[i] < nums2[j]:
                nums1[i+j] = nums1_copy[i]
                i += 1
            else:
                nums1[i+j] = nums2[j]
                j += 1

class Solution {
public:
    void merge(vector<int>& nums1, int m, vector<int>& nums2, int n) {
        int cur_a = m-1, cur_b = n-1, cur_r = m+n-1;
        while (cur_a >= 0 && cur_b >= 0) {
            if (nums1[cur_a] > nums2[cur_b]) {
                nums1[cur_r--] = nums1[cur_a--];
            } else {
                nums1[cur_r--] = nums2[cur_b--];
            }
        }
        while (cur_b >= 0) {
            nums1[cur_r--] = nums2[cur_b--];
        }

    }

2 合并两个有序链表

分析

采用双指针法，将两个链表的元素按顺序合并。

  • Python
  • C++
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def mergeTwoLists(self, list1: Optional[ListNode], list2: Optional[ListNode]) -> Optional[ListNode]:
        cur_a = list1; cur_b = list2; dummy = ListNode(); result_cur = dummy 
        while cur_a and cur_b:
            if cur_a.val < cur_b.val:
                result_cur.next = cur_a
                result_cur = result_cur.next
                cur_a = cur_a.next
            else:
                result_cur.next = cur_b
                result_cur = result_cur.next
                cur_b = cur_b.next
        result_cur.next = cur_a if not cur_b else cur_b
        return dummy.next
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* mergeTwoLists(ListNode* list1, ListNode* list2) {
        ListNode *a = list1, *b = list2;
        ListNode dummy(-1);
        ListNode *p = &dummy;
        while (a && b) {
            if (a->val < b->val) {
                p->next = a;
                p = p->next;
                a = a->next;
            } else {
                p->next = b;
                p = p->next;
                b = b->next;
            }
        }
        p->next = !b ? a : b;
        return dummy.next;
    }
};

# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def mergeKLists(self, lists: List[Optional[ListNode]]) -> Optional[ListNode]:
        result = None
        for i in range(len(lists)):
            list2 = lists[i]
            result = self.merge2Lists(result, list2)
        return result
    
    def merge2Lists(self, list1: Optional[ListNode], list2: Optional[ListNode]) -> Optional[ListNode]:
        a = list1; b = list2
        dummy = ListNode(); p = dummy
        while a and b:
            if a.val < b.val:
                p.next = a
                a = a.next
            else:
                p.next = b
                b = b.next
            p = p.next
        p.next = a if not b else b
        return dummy.next
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* mergeKLists(vector<ListNode*>& lists) {
        ListNode *result = nullptr;
        for (auto list2: lists) {
            result = merge2Lists(result, list2);
        }
        return result;
    }

    ListNode* merge2Lists(ListNode *list1, ListNode *list2) {

        ListNode *p = &dummy;
        while (list1 && list2) {
            if (list1->val < list2->val) {
                p->next = list1;
                list1 = list1->next;
            } else {
                p->next = list2;
                list2 = list2->next;
            }
            p = p->next;
        }
        p->next = !list1 ? list2 : list1;
        return dummy.next;
    }

4 对链表进行插入排序

分析

采用双指针法，将链表分成已排序和未排序两部分，将未排序的元素插入到已排序的元素中。

  • Python
  • C++
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def insertionSortList(self, head: Optional[ListNode]) -> Optional[ListNode]:
        cur = head
        dummy = ListNode(val=-5001, next=head)
        tail = dummy
        while cur:
            if cur.val >= tail.val:
                tail.next = cur
                cur = cur.next
                tail = tail.next
            else:
                cur_i = dummy
                while cur_i.val < cur.val and cur_i.next.val < cur.val:
                    cur_i = cur_i.next
                tail.next = cur.next
                cur.next = cur_i.next
                cur_i.next = cur
                cur = tail.next
        return dummy.next
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* insertionSortList(ListNode* head) {
        ListNode *cur = head;
        ListNode dummy(-5001, head);
        ListNode *tail = &dummy;
        while (cur) {
            if (cur->val >= tail->val) {
                tail->next = cur;
                cur = cur->next;
                tail = tail->next;
            } else {
                ListNode *cur_i = &dummy;
                while (cur_i->val < cur->val and cur_i->next->val < cur->val) {
                    cur_i = cur_i->next;
                }
                tail->next = cur->next;
                cur->next = cur_i->next;
                cur_i->next = cur;
                cur = tail->next;
            }
        }
        return dummy.next;
    }
};

5 排序链表

分析

采用归并排序，将链表分成两部分，分别排序，然后合并。

  • Python
  • C++
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def sortList(self, head: Optional[ListNode]) -> Optional[ListNode]:
        if not head or not head.next:
            return head
        dummy = ListNode(next=head)
        slow = dummy
        fast = dummy
        while fast and fast.next:
            slow = slow.next
            fast = fast.next.next
        mid = slow.next
        slow.next = None  # 需要将节点进行断开
        left = self.sortList(head)
        right = self.sortList(mid)
        return self.merge(left, right)
        


    def merge(self, list1: Optional[ListNode], list2: Optional[ListNode]) -> Optional[ListNode]:
        dummy = ListNode()
        cur = dummy
        while list1 and list2:
            if list1.val < list2.val:
                cur.next = list1
                list1 = list1.next
            else:
                cur.next = list2
                list2 = list2.next
            cur = cur.next
        cur.next = list1 if not list2 else list2
        return dummy.next
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* sortList(ListNode* head) {
        if (!head || !head->next) return head;
        ListNode dummy(-1, head);
        ListNode *slow = &dummy, *fast = &dummy;
        while (fast && fast->next) {
            slow = slow->next;
            fast = fast->next->next;
        }
        ListNode *mid = slow->next;
        slow->next = nullptr;
        ListNode *left = sortList(head);
        ListNode *right = sortList(mid);
        return merge(left, right);
    }

    ListNode* merge(ListNode *list1, ListNode *list2) {
        ListNode dummy(-1);
        ListNode *cur = &dummy;
        while (list1 && list2) {
            if (list1->val < list2->val) {
                cur->next = list1;
                list1 = list1->next;
            } else {
                cur->next = list2;
                list2 = list2->next;
            }
            cur = cur->next;
        }
        cur->next = !list2 ? list1 : list2;
        return dummy.next;
    }
};

6 缺失的第一个正整数

分析

采用原地哈希法，将数组中的元素放到其应该在的位置上。

这道题的关键在于： 1. 根据题目要求，我们需要找的是缺失的第一个正整数，且要求O(n)时间复杂度和O(1)空间复杂度。 2. 对于长度为n的数组，如果数组包含了所有1到n的正整数，那么缺失的第一个正整数就是n+1。 3. 如果数组中缺少了某个1到n之间的正整数，那么缺失的第一个正整数就是其中最小的那个。

原地哈希法的核心思想是： - 利用数组的索引作为哈希表的键，将数值放到对应的位置上。 - 对于值为i的元素，应该放在索引i-1的位置上（因为数组索引从0开始）。

具体实现步骤： 1. 首先将所有不在[1, n]范围内的数标记为n+1，因为这些数不会是答案。 2. 然后遍历数组，对于每个在[1, n]范围内的数x，将nums[x-1]标记为负数，表示数字x已经出现过。 3. 最后再次遍历数组，找到第一个正数的位置i，则i+1就是缺失的第一个正整数。 4. 如果所有位置都是负数，说明1到n都出现过，返回n+1。

这种方法巧妙地利用了数组本身作为哈希表，通过正负号标记元素是否出现，从而实现了O(1)的额外空间复杂度。

  • Python
  • C++
class Solution:
    def firstMissingPositive(self, nums: List[int]) -> int:
        n = len(nums)
        # 第一步：将所有不在 [1, n] 范围内的数标记为 n+1
        for i in range(n):
            if nums[i] <= 0 or nums[i] > n:
                nums[i] = n + 1
        # 第二步：使用索引作为哈希键，将出现过的数对应的位置标记为负数
        for i in range(n):
            num = abs(nums[i])
            if 1 <= num <= n and nums[num-1]>0:
                # 防止重复取反
                nums[num - 1] = -nums[num - 1]
        # 第三步：找到第一个正数所在的位置，返回其索引加一
        for i in range(n):
            if nums[i] > 0:
                return i + 1
        # 如果所有位置都是负数，说明 1 到 n 都出现过，返回 n+1
        return n + 1
#include<cmath>
class Solution {
public:
    int firstMissingPositive(vector<int>& nums) {
        int n = nums.size();
        for (int i = 0; i < n; i++) {
            if (nums[i] <= 0 || nums[i] > n) {
                nums[i] = n + 1;
            }
        }
        for ( int i = 0; i < n; i++) {
            int num = abs(nums[i]);
            if (1 <= num && num <= n && nums[num-1]>0) {
                nums[num-1] = -nums[num-1];
            }
        }
        for (int i = 0; i < n; i++) {
            if (nums[i] > 0) {
                return i + 1;
            }
        }
        return n + 1;
    }
};

7 颜色分类

分析

采用双指针法，将数组分成三部分，分别表示0、1、2。

  • Python
  • C++
class Solution:
    def sortColors(self, nums: List[int]) -> None:
        """
        Do not return anything, modify nums in-place instead.
        """
        p0 = 0          # 指向当前已排序的0的末尾的下一个位置
        p2 = len(nums) - 1  # 指向当前已排序的2的开头的前一个位置
        i = 0           # 当前遍历的指针
        while i <= p2:
            if nums[i] == 0:
                # 将0交换到前面
                nums[i], nums[p0] = nums[p0], nums[i]
                p0 += 1
                i += 1  # 只有当交换的是0时，i才增加，因为交换过来的nums[p0-1]已经是处理过的0
            elif nums[i] == 2:
                # 将2交换到后面
                nums[i], nums[p2] = nums[p2], nums[i]
                p2 -= 1
                # 注意：这里不增加i，因为交换过来的nums[i]可能是0或1，需要再次检查
            else:
                # nums[i] == 1，无需交换，只需移动指针
                i += 1
class Solution {
public:
    void sortColors(vector<int>& nums) {
        int p0 = 0, p2 = nums.size() - 1, i = 0;
        while (i <= p2) {
            if (nums[i] == 0) {
                swap(nums[i++], nums[p0++]);
            } else if (nums[i] == 2) {
                swap(nums[i], nums[p2--]);
            } else {
                i += 1;
            }
        }      
    }
};
