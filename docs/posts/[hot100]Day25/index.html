3 二叉树的锯齿形层序遍历

分析

在层序遍历的基础上，用一个变量存储输出结果的方向，若表示从右往左，则可在形成列表元素后翻转再填入结果列表中即可。

  • Python
  • C++
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def zigzagLevelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        from collections import deque
        if not root: return []
        ans = []
        q = deque([root])
        direction = 0
        while q:
            vals = []
            for _ in range(len(q)):
                cur = q.popleft()
                vals.append(cur.val)
                if cur.left:
                    q.append(cur.left)
                if cur.right:
                    q.append(cur.right)
            if direction % 2 == 0:
                ans.append(vals)
            else:
                ans.append(vals[::-1])
            direction += 1
        return ans
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
#include <queue>
#include <algorithm>
// 广度优先
class Solution {
public:
    vector<vector<int>> zigzagLevelOrder(TreeNode* root) {
        vector<vector<int>> ans;
        if (!root) {
            return ans;
        }   
        queue<TreeNode*> q;
        q.push(root);
        int direction = 0;
        while (!q.empty()) {
            vector<int> vals;
            int num = q.size();
            for (int i = 0; i < num; i++) {  // 此处的i < num是动态比较的如果采用i < q.size()会造成问题这python中不同
                vals.push_back(q.front()->val);
                if (q.front()->left) q.push(q.front()->left);
                if (q.front()->right) q.push(q.front()->right);
                q.pop();
            }
            if (direction % 2 == 0) ans.push_back(vals);
            else {
                reverse(vals.begin(), vals.end());
                ans.push_back(vals);
            }
            direction += 1;
        }
        return ans;
    }
};

4 恢复二叉搜索树

分析

恢复二叉树指的的是节点左子树元素都比节点小，右子树节点都比节点大，在中序遍历的情况下，搜索树的遍历结果呈现递增的情况，故只需要找到第一个违反递增规律的前节点和第二个违反规律的后节点，交换二者位置即可（采用morris中序遍历）。

  • Python
  • C++
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def recoverTree(self, root: Optional[TreeNode]) -> None:
        """
        Do not return anything, modify root in-place instead.
        """
        # Morris 中序遍历
        cur = root
        firstNode = None
        secondNode = None
        pre = None

        while cur:
            if cur.left:
                # 找到左子树的最右节点
                mostright = cur.left
                while mostright.right and mostright.right != cur:
                    mostright = mostright.right
                
                if not mostright.right:
                    # 建立线索，继续向左
                    mostright.right = cur
                    cur = cur.left
                else:
                    # 恢复线索，访问当前节点
                    mostright.right = None
                    if pre and pre.val > cur.val:
                        if not firstNode:
                            firstNode = pre
                        secondNode = cur
                    pre = cur
                    cur = cur.right
            else:
                # 没有左子树，直接访问当前节点
                if pre and pre.val > cur.val:
                    if not firstNode:
                        firstNode = pre
                    secondNode = cur
                pre = cur
                cur = cur.right

        # 交换错误节点的值
        if firstNode and secondNode:
            firstNode.val, secondNode.val = secondNode.val, firstNode.val
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    void recoverTree(TreeNode* root) {
        TreeNode *cur = root, *firstNode = nullptr, *secondNode = nullptr, *pre = nullptr;
        while (cur) {
            if (cur->left) {
                TreeNode *mostright = cur->left;
                while (mostright->right && mostright->right != cur) {
                    mostright = mostright->right;
                }
                if (!mostright->right) {
                    mostright->right = cur;
                    cur = cur->left;
                } else {
                    mostright->right = nullptr;
                    if (pre && pre->val > cur->val) {
                        if (!firstNode) firstNode = pre;
                        secondNode = cur;
                    }
                    pre = cur;
                    cur = cur->right;
                }
            } else {
                if (pre && pre->val > cur->val) {
                    if (!firstNode) firstNode = pre;
                    secondNode = cur;
                }
                pre = cur;
                cur = cur->right;
            }
        }
        if (firstNode && secondNode) {
                int temp = firstNode->val;
                firstNode->val = secondNode->val;
                secondNode->val = temp;
            }
    }
};

5 相同的树

分析

采用递归法，不断递归节点左子节点和右子节点是否相同。

  • Python
  • C++
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def isSameTree(self, p: Optional[TreeNode], q: Optional[TreeNode]) -> bool:
        if not p and not q: return True
        if not p or not q: return False
        return p.val == q.val and self.isSameTree(p.left, q.left) and self.isSameTree(p.right, q.right)
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    bool isSameTree(TreeNode* p, TreeNode* q) {
        if (!p && !q) return true;
        if (!p || !q) return false;
        return p->val == q->val && isSameTree(p->left, q->left) && isSameTree(p->right, q->right);
    }
};
